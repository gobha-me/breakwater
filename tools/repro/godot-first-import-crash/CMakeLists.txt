cmake_minimum_required(VERSION 3.28)
project(repro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
FetchContent_Declare(godot-cpp GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git GIT_TAG 10.0.0-stable GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(godot-cpp)
add_library(repro SHARED src/repro.cpp)
target_link_libraries(repro PRIVATE godot-cpp)
set_target_properties(repro PROPERTIES PREFIX "lib" OUTPUT_NAME "repro" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/project/bin")
